import { pgTable, text, timestamp, integer, boolean, uuid, jsonb } from "drizzle-orm/pg-core";

export const categories = pgTable("categories", {
  id: uuid("id").defaultRandom().primaryKey(),
  name: text("name").notNull(),
  slug: text("slug").notNull().unique(),
  sku: text("sku").unique(),
  createdAt: timestamp("created_at").defaultNow().notNull(),
});

export const products = pgTable("products", {
  id: uuid("id").defaultRandom().primaryKey(),
  categoryId: uuid("category_id").references(() => categories.id),
  name: text("name").notNull(),
  slug: text("slug").notNull().unique(),
  sku: text("sku").unique(),
  description: text("description"),
  price: integer("price").notNull().default(0),
  compareAtPrice: integer("compare_at_price"), 
  stock: integer("stock").notNull().default(0),
  status: text("status").notNull().default("draft"),
  isFeatured: boolean("is_featured").notNull().default(false),
  mainImage: text("main_image"),
  gallery: jsonb("gallery").$type<string[]>().default([]),
  specifications: jsonb("specifications").$type<{key: string, value: string}[]>().default([]),
  ingredients: text("ingredients"),
  usage: text("usage"),
  createdAt: timestamp("created_at").defaultNow().notNull(),
  updatedAt: timestamp("updated_at").defaultNow().notNull(),
});

export const adminUsers = pgTable("admin_users", {
  id: uuid("id").defaultRandom().primaryKey(),
  email: text("email").notNull().unique(),
  passwordHash: text("password_hash").notNull(),
  role: text("role").notNull().default("admin"),
  createdAt: timestamp("created_at").defaultNow().notNull(),
});

export const siteSettings = pgTable("site_settings", {
  key: text("key").primaryKey(),
  value: jsonb("value").notNull(),
  updatedAt: timestamp("updated_at").defaultNow().notNull(),
});
export const orders = pgTable("orders", {
  id: uuid("id").defaultRandom().primaryKey(),
  firstName: text("first_name").notNull(),
  lastName: text("last_name").notNull(),
  phone: text("phone").notNull(),
  address: text("address").notNull(),
  city: text("city").notNull(),
  postalCode: text("postal_code").notNull(),
  notes: text("notes"),
  totalAmount: integer("total_amount").notNull(),
  status: text("status").notNull().default("pending"), // pending, paid, processing, shipped, delivered, cancelled
  trackingCode: text("tracking_code").notNull().unique().default("FL-PENDING"),
  createdAt: timestamp("created_at").defaultNow().notNull(),
});

export const orderItems = pgTable("order_items", {
  id: uuid("id").defaultRandom().primaryKey(),
  orderId: uuid("order_id").references(() => orders.id).notNull(),
  productId: uuid("product_id").references(() => products.id).notNull(),
  productName: text("product_name").notNull(),
  price: integer("price").notNull(),
  quantity: integer("quantity").notNull(),
});
export const articles = pgTable("articles", {
  id: uuid("id").defaultRandom().primaryKey(),
  title: text("title").notNull(),
  slug: text("slug").notNull().unique(),
  excerpt: text("excerpt"),
  content: text("content").notNull(),
  category: text("category").notNull().default("education"),
  coverImage: text("cover_image"),
  status: text("status").notNull().default("draft"),
  readTime: text("read_time").default("۵ دقیقه"),
  author: text("author").default("تیم فاطمه‌لب"),
  featured: boolean("featured").notNull().default(false),

export default async function ProductsPage({ searchParams }: { searchParams: Promise<{ q?: string; category?: string }> }) {
  const { q = '', category = '' } = await searchParams;
  const searchTerm = q.trim();

  const categoryRows = await db.select().from(categories).orderBy(categories.name);
  const selectedCategory = categoryRows.find((item) => item.slug === category);
  const invalidCategory = Boolean(category && !selectedCategory);

  const conditions: SQL[] = [eq(products.status, 'published')];
  if (searchTerm) conditions.push(ilike(products.name, `%${searchTerm}%`));
  if (selectedCategory) conditions.push(eq(products.categoryId, selectedCategory.id));

  const dbProducts = invalidCategory ? [] : await db.select().from(products).where(and(...conditions)).orderBy(desc(products.createdAt));

  const buildCategoryHref = (slug?: string) => {
    const params = new URLSearchParams();
    if (searchTerm) params.set('q', searchTerm);
    if (slug) params.set('category', slug);
    const query = params.toString();
    return query ? `/products?${query}` : '/products';
  };

  return (
    <div className="min-h-screen flex flex-col bg-background">
      <Header />
      <main className="flex-1">
        <section className="fl-page-intro">
          <div className="container-custom">

              <Search className="pointer-events-none absolute right-3.5 top-1/2 -translate-y-1/2 text-[#99919d]" size={17} />
              <input type="search" name="q" defaultValue={searchTerm} placeholder="نام محصول را جست‌وجو کنید..." className="fl-input pr-10 pl-4" aria-label="جست‌وجوی محصولات" />

export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
  const { slug } = await params;
  const [product] = await db.select().from(products).where(eq(products.slug, slug)).limit(1);
  if (!product || product.status !== 'published') return { title: 'محصول یافت نشد' };
  return {
    title: product.name,
    description: product.description || `مشاهده مشخصات و اطلاعات ${product.name} در فاطمه‌لب.`,
  };
}

'use client';

import { useCartStore } from '@/store/cart';
import { ShoppingBag } from 'lucide-react';
import { toast } from 'sonner';

type CartProduct = {
  id: string;
  name: string;
  slug: string;
  price: number;
  stock: number;
  mainImage: string | null;
};

export default function AddToCartButton({ product }: { product: CartProduct }) {
  const addItem = useCartStore((state) => state.addItem);
  const hasPrice = product.price > 0;
  const hasStock = product.stock > 0;
  const canPurchase = hasPrice && hasStock;

  const handleAdd = () => {
    if (!hasPrice) { toast.error('این محصول هنوز قیمت‌گذاری نشده است.'); return; }
    if (!hasStock) { toast.error('این محصول در حال حاضر موجود نیست.'); return; }
    addItem({ id: product.id, name: product.name, slug: product.slug, price: product.price, mainImage: product.mainImage });
    toast.success('محصول به سبد خرید اضافه شد.');
  };

  return (
    <div className="space-y-2">
      <button 
        type="button" 
        onClick={handleAdd} 
        disabled={!canPurchase} 
        className="btn-primary min-h-[48px] w-full disabled:cursor-not-allowed disabled:border-[#ddd7de] disabled:bg-[#efedf0] disabled:text-[#8f8791] disabled:shadow-none"
      >
        <ShoppingBag size={17} />
        {canPurchase ? 'افزودن به سبد خرید' : !hasPrice ? 'به زودی عرضه می‌شود' : 'در حال حاضر ناموجود'}
      </button>
      {!canPurchase && (
        <p className="text-center text-[10px] leading-6 text-[#8b818c]">
          {hasPrice ? 'این محصول در حال حاضر قابل سفارش نیست.' : 'امکان سفارش این محصول پس از ثبت قیمت فعال می‌شود.'}
        </p>
      )}
    </div>
  );
}

export async function getCachedPublishedProducts() {

const unverifiedSeedDescriptions = new Set([
  'آبرسان و شفاف‌کننده پوست',
  'تقویت‌کننده و ضخیم‌کننده مو',
  'پاک‌سازی عمیق پوست',
  'پوشش طبیعی و ماندگار',
]);

const knownNonProductAssets = [
  'home-services.webp',
  'lumen.png.png',
];

export function getPublicProductDescription(description: string | null | undefined): string | null {
  const value = description?.replace(/\s+/g, ' ').trim();
  if (!value) return null;
  if (unverifiedSeedDescriptions.has(value)) return null;
  return value;
}

export function getPublicProductImage(image: string | null | undefined): string | null {
  if (!image) return null;
  const normalized = image.toLowerCase().split('?')[0];
  if (knownNonProductAssets.some((asset) => normalized.includes(asset))) return null;
  return image;
}
